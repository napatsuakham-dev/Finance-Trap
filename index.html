function getTimeStr() {

function selectChoice(text, choiceIndex) {

  document.getElementById('choicesArea').style.display = 'none';

  document.getElementById('choicesArea').innerHTML = '';


  // Track choice with dynamic points - get points for this specific choice in this specific question

  const config = getConfig();

  const level = parseInt(config.current_level || '1');

  const levelPoints = choicePoints[level] || [];

  const questionPoints = levelPoints[currentTurn] || [0, 0, 0, 0];  // [A, B, C, D]

  const points = questionPoints[choiceIndex] || 0;  // Get score for selected choice

  

  choiceHistory.push({ choiceIndex, points });


  // Add time stamp

  const cb = document.getElementById('chatBody');

  const timeDiv = document.createElement('div');

  timeDiv.style.cssText = 'text-align:center;font-size:11px;color:rgba(255,255,255,0.6);margin:8px 0';

  timeDiv.textContent = getTimeStr();

  cb.appendChild(timeDiv);


  addBubble('player', text);

  playSound('notification');


  currentTurn++;

  const allTurns = getTurnsForCurrentLevel();

  if (currentTurn < allTurns.length) {

    setTimeout(() => playTurn(currentTurn), 800);

  } else {

    // End of chapter with analysis

    setTimeout(() => showEndAnalysis(), 1000);

  }

}

async function playTurn(turnIdx) {

  const allTurns = getTurnsForCurrentLevel();

  const turn = allTurns[turnIdx];

  const msgs = turn.messages;


  for (let i = 0; i < msgs.length; i++) {

    const msg = msgs[i];

    const delay = msg.delay || (i === 0 ? 600 : 1000);


    if (msg.type === 'officer' || msg.type === 'jen' || msg.type === 'mint' || msg.type === 'friend' || msg.type === 'admin' || msg.type === 'sticker' || msg.type === 'image') {

      showTyping();

      await sleep(delay);

      hideTyping();

    }


    if (msg.type === 'jen' || msg.type === 'officer' || msg.type === 'mint' || msg.type === 'friend' || msg.type === 'admin') {

      addBubble(msg.type, msg.text());

    } else if (msg.type === 'sticker') {

      addBubble('sticker', msg.emoji);

    } else if (msg.type === 'image') {

      addBubble('image', `<div style="font-size:28px">${msg.label.substring(0, 2)}</div><div style="font-size:12px;font-weight:600">${msg.label.substring(2)}</div><div style="font-size:11px;color:#888">${msg.sublabel}</div>`);

    }


    await sleep(300);

  }


  // Add read timestamp

  const cb = document.getElementById('chatBody');

  const timeDiv = document.createElement('div');

  timeDiv.style.cssText = 'text-align:center;font-size:11px;color:rgba(255,255,255,0.6);margin:8px 0';

  timeDiv.textContent = getTimeStr();

  cb.appendChild(timeDiv);


  showChoices(turn.choices);

}

}
